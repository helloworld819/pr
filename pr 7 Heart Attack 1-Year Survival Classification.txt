# 1. Load dataset & Missing Values Inspection
names = c("survival", "still_alive", "age", "pericardial_effusion", "fractional_shortening", "epss", "lvdd", "wall_motion_score", "wall_motion_index", "mult", "name", "group", "alive_at_1")
raw_echo = read_csv("echocardiogram.data", col_names = names, na = c("?", "NA", ""), show_col_types = FALSE)

na_counts = colSums(is.na(raw_echo))
print("Missing values per column:")
print(na_counts)

missing_df = data.frame(
  Feature = factor(names(raw_echo), levels = rev(names(raw_echo))),
  Missing_Count = as.numeric(na_counts),
  Missing_Pct = as.numeric(round(colMeans(is.na(raw_echo)) * 100, 1))
)

# Plot missing values count & percentage by feature (ggplot2)
ggplot(missing_df, aes(x = reorder(Feature, Missing_Count), y = Missing_Count, fill = Missing_Count)) +
  geom_col() +
  geom_text(aes(label = paste0(Missing_Count, " (", Missing_Pct, "%)")), hjust = -0.1, size = 3.5) +
  coord_flip() +
  scale_fill_gradient(low = "steelblue", high = "firebrick") +
  scale_y_continuous(expand = expansion(mult = c(0, 0.15))) +
  labs(
    title = "Missing Values Count & Percentage by Feature",
    subtitle = paste0("Dataset: echocardiogram.data (Total Rows = ", nrow(raw_echo), ")"),
    x = "Feature Name",
    y = "Missing Values Count"
  ) +
  theme_minimal() +
  theme(legend.position = "none", plot.title = element_text(face = "bold"))

2. Data Cleaning, Target Definition & Survival Distribution
echo = raw_echo %>%
  mutate(
    survive_1year = case_when(
      alive_at_1 == "1" ~ 1, alive_at_1 == "0" ~ 0, survival >= 12 ~ 1, survival < 12 & still_alive == 0 ~ 0, TRUE ~ NA_real_
    )
  ) %>%
  filter(!is.na(survive_1year)) %>%
  select(age, pericardial_effusion, fractional_shortening, epss, lvdd, wall_motion_index, survive_1year) %>%
  mutate(across(c(age, fractional_shortening, epss, lvdd, wall_motion_index), ~ ifelse(is.na(.x), median(.x, na.rm = TRUE), .x))) %>%
  mutate(survive_1year = factor(survive_1year, levels = c(0, 1), labels = c("Died", "Survived")))

ggplot(echo, aes(x = survive_1year, fill = survive_1year)) +
  geom_bar(show.legend = FALSE) +
  scale_fill_manual(values = c("Died" = "lightcoral", "Survived" = "lightgreen")) +
  labs(title = "1-Year Patient Survival Distribution", x = "Outcome", y = "Patient Count") +
  theme_minimal()

3. Train Test Split & Logistic Regression Model
set.seed(42)
train_idx = sample(seq_len(nrow(echo)), size = 0.75 * nrow(echo))
train = echo[train_idx, ]
test = echo[-train_idx, ]

logit_model = glm(survive_1year ~ age + fractional_shortening + epss + lvdd + wall_motion_index, data = train, family = binomial)
summary(logit_model)

logit_prob = predict(logit_model, newdata = test, type = "response")
logit_pred = factor(ifelse(logit_prob > 0.5, "Survived", "Died"), levels = c("Died", "Survived"))

logit_cm = table(Actual = test$survive_1year, Predicted = logit_pred)
logit_cm
logit_acc = sum(diag(logit_cm)) / sum(logit_cm)
logit_acc

4. Tuned Decision Tree Risk Modeling (rpart & rpart.plot)
# Hyperparameter regularization to prevent overfitting on small clinical sample
dt_model = rpart(
  survive_1year ~ .,
  data = train,
  method = "class",
  control = rpart.control(cp = 0.001, maxdepth = 2, minsplit = 4)
)
rpart.plot(
  dt_model,
  type = 2,
  extra = 104,
  under = TRUE,
  fallen.leaves = TRUE,
  box.palette = "RdYlGn",
  main = "Tuned Decision Tree for 1-Year Heart Attack Survival"
)

 5. Tuned Decision Tree Evaluation & Actual vs Predicted Plot
dt_pred = predict(dt_model, newdata = test, type = "class")
dt_cm = table(Actual = test$survive_1year, Predicted = dt_pred)
dt_cm
dt_acc = sum(diag(dt_cm)) / sum(dt_cm)
dt_acc

eval_df = data.frame(Actual = test$survive_1year, Predicted = dt_pred)
ggplot(eval_df, aes(x = Actual, fill = Predicted)) +
